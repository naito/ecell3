143/**
144 Defines methods for the class stub.
145
146 This macro does two things:
147 (1) template specialization of PropertyInterface for CLASSNAME
148 is necessary at this point to instantiate thePropertySlotMap static
149 variable.
150 (2) thePropertyInterface static variable must also be instantiated.
151
152 @param CLASSNAME the name of the EcsObject class
153 @param DMTYPE the root class of the class to declare.
154 (Process, Variable, System, Stepper)
155 */
156#define LIBECS_DM_INIT_STATIC( CLASSNAME, DMTYPE ) \
157 template class libecs::PropertyInterface< CLASSNAME >;\
158 char CLASSNAME::thePropertyInterface[ sizeof( libecs::PropertyInterface<CLASSNAME> ) ]; \
159 char CLASSNAME::theDMTypeName[] = #DMTYPE; \
160 libecs::PropertyInterface< CLASSNAME >& CLASSNAME::_getPropertyInterface() \
161 { \
162 return *reinterpret_cast< libecs::PropertyInterface< CLASSNAME >* >( thePropertyInterface ); \
163 } \
164 void CLASSNAME::initializeModule() \
165 { \
166 new(thePropertyInterface) libecs::PropertyInterface<CLASSNAME>( #CLASSNAME, theDMTypeName ); \
167 } \
168 void CLASSNAME::finalizeModule() \
169 { \
170 reinterpret_cast< libecs::PropertyInterface<CLASSNAME>* >( thePropertyInterface )->~PropertyInterface<CLASSNAME>(); \
171 } \
172 libecs::PropertySlotBase const* CLASSNAME::getPropertySlot( libecs::String const& aPropertyName ) const \
173 { \
174 return _getPropertyInterface().getPropertySlot( aPropertyName ); \
175 } \
176 void CLASSNAME::setProperty( libecs::String const& aPropertyName, libecs::Polymorph const& aValue ) \
177 { \
178 return _getPropertyInterface().setProperty( *this, aPropertyName, aValue ); \
179 } \
180 libecs::Polymorph CLASSNAME::getProperty( libecs::String const& aPropertyName ) const \
181 { \
182 return _getPropertyInterface().getProperty( *this, aPropertyName ); \
183 } \
184 void CLASSNAME::loadProperty( libecs::String const& aPropertyName, libecs::Polymorph const& aValue ) \
185 { \
186 return _getPropertyInterface().loadProperty( *this, aPropertyName, aValue ); \
187 } \
188 libecs::Polymorph CLASSNAME::saveProperty( libecs::String const& aPropertyName ) const \
189 { \
190 return _getPropertyInterface().saveProperty( *this, aPropertyName ); \
191 } \
192 std::vector< libecs::String > CLASSNAME::getPropertyList() const \
193 { \
194 return _getPropertyInterface().getPropertyList( *this ); \
195 } \
196 libecs::PropertySlotProxy* CLASSNAME::createPropertySlotProxy( libecs::String const& aPropertyName ) \
197 { \
198 return _getPropertyInterface().createPropertySlotProxy( *this, aPropertyName ); \
199 } \
200 libecs::PropertyAttributes \
201 CLASSNAME::getPropertyAttributes( libecs::String const& aPropertyName ) const \
202 { \
203 return _getPropertyInterface().getPropertyAttributes( *this, aPropertyName ); \
204 } \
205 libecs::PropertyInterfaceBase const& CLASSNAME::getPropertyInterface() const \
206 {\
207 return _getPropertyInterface(); \
208 } \
209 DynamicModuleInfo const* CLASSNAME::getClassInfoPtr()\
210 {\
211 return static_cast<const DynamicModuleInfo*>( &_getPropertyInterface() );\
212 }

594/**
595 Common base class for classes with PropertySlots.
596
597 @see PropertySlot
598*/
599
600class LIBECS_API EcsObject
601{
602public:
603
604 LIBECS_DM_INIT_PROP_INTERFACE()
605 {
606 ; // empty, but this must be here.
607 }
608
609
610 EcsObject()
611 : theModel( 0 ), theHandle( Handle::INVALID_HANDLE_VALUE ),
612 disposed_( false )
613 {
614 ; // do nothing
615 }
616
617 virtual ~EcsObject()
618 {
619 dispose();
620 }
621
622 virtual void dispose();
623
624 virtual void detach();
625
626 /**
627 Get a Model object to which this object belongs.
628
629 @return a borrowed pointer to the Model.
630 */
631 Model* getModel() const
632 {
633 return theModel;
634 }
635
636 /**
637 Associate a Model object with this object.
638
639 @internal
640 @param aModel the Model object to associate with this object.
641 */
642 void setModel( Model* const aModel )
643 {
644 theModel = aModel;
645 }
646
647 /**
648 Return true if the object is already disposed.
649 @return true if the object is disposed, false otherwise.
650 */
651 bool isDisposed() const
652 {
653 return disposed_;
654 }
655
656 /**
657 Return the PropertySlot for aPropertyName
658
659 @param aPropertyName the name of the property.
660 @return the pointer to the PropertySlot.
661 */
662 virtual PropertySlotBase const*
663 getPropertySlot( String const& aPropertyName ) const = 0;
664
665
666 /**
667 Set the value of the property specified by aPropertyName
668 to aValue.
669
670 @param aPropertyName the name of the property.
671 @param aValue the value to set the property to.
672 @return the value of the property.
673 */
674 virtual void
675 setProperty( String const& aPropertyName, Polymorph const& aValue ) = 0;
676
677
678 /**
679 Return the Polymorph'ed value for aPropertyName
680
681 @param aPropertyName the name of the property.
682 @return the value of the property.
683 */
684 virtual Polymorph
685 getProperty( String const& aPropertyName ) const = 0;
686
687 virtual void
688 loadProperty( String const& aPropertyName, Polymorph const& aValue ) = 0;
689
690 virtual Polymorph saveProperty( String const& aPropertyName ) const = 0;
691
692 virtual std::vector< String > getPropertyList() const = 0;
693
694 virtual PropertyAttributes
695 getPropertyAttributes( String const& aPropertyName ) const = 0;
696
697 virtual PropertyInterfaceBase const& getPropertyInterface() const = 0;
698 virtual void defaultSetProperty( String const& aPropertyName,
699 Polymorph const& aValue );
700
701 virtual Polymorph defaultGetProperty( String const& aPorpertyName ) const;
702
703 virtual std::vector< String > defaultGetPropertyList() const;
704
705 virtual PropertyAttributes
706 defaultGetPropertyAttributes( String const& aPropertyName ) const;
707
708 LIBECS_DEPRECATED String const& getClassName() const;
709
710 virtual String asString() const;
711
712 Handle const& getHandle() const
713 {
714 return theHandle;
715 }
716
717 /// @internal
718 void setHandle( Handle const& aHandle )
719 {
720 theHandle = aHandle;
721 }
722
723 /// @internal
724 template <typename Type>
725 void nullSet( typename Param< Type >::type );
726
727 /// @internal
728 template <typename Type>
729 Type nullGet() const;
730
731private:
732
733 void throwNotSetable() const;
734 void throwNotGetable() const;
735
736protected:
737 Model* theModel;
738 Handle theHandle;
739 bool disposed_;
740};
741
742template <typename Type>
743inline void EcsObject::nullSet( typename Param< Type >::type )
744{
745 throwNotSetable();
746}
747
748template <typename Type>
749inline Type EcsObject::nullGet() const
750{
751 throwNotGetable();
752}

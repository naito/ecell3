71/// Base exception class
72class LIBECS_API Exception: public std::exception
73{
74public:
75 /**
76 Constructor.
77 @param method the name of the method from where the exception is thrown.
78 @param message the description of the exception.
79 @param entity the entity where the exception occurred.
80 */
81 Exception( String const& method, String const& message = "", EcsObject const* object = 0 );
82
83 virtual ~Exception() throw();
84
85 const String& message() const throw()
86 {
87 return theMessage;
88 }
89
90 virtual const char* what() const throw();
91
92 virtual const char* getClassName() const throw()
93 {
94 return "Exception";
95 }
96
97 virtual String const& getMethod() const throw()
98 {
99 return theMethod;
100 }
101
102 EcsObject const* getEcsObject() const throw();
103
104protected:
105 const String theMethod;
106 const String theMessage;
107 Handle const theEcsObjectHandle;
108 Model const* const theModel;
109 mutable String theWhatMsg;
110};
